local dialogueLayoutOrder = 0

local function speakDialogue(text)
    if not dialogueContainer then return end

    dialogueLayoutOrder += 1

    local lineFrame = Instance.new("Frame")
    lineFrame.Name = "LineFrame"
    lineFrame.Size = UDim2.new(1, 0, 0, 36)
    lineFrame.BackgroundTransparency = 1
    lineFrame.LayoutOrder = dialogueLayoutOrder
    lineFrame.Parent = dialogueContainer

    local sliceCount = 6
    local sliceHeightRatio = 1 / sliceCount
    local labels = {}
    local sliceFrames = {}

    for i = 1, sliceCount do
        local sliceFrame = Instance.new("Frame")
        sliceFrame.Size = UDim2.new(1, 0, sliceHeightRatio, 0)
        sliceFrame.Position = UDim2.new(0, 0, (i - 1) * sliceHeightRatio, 0)
        sliceFrame.ClipsDescendants = true
        sliceFrame.BackgroundTransparency = 1
        sliceFrame.Parent = lineFrame

        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, 0, sliceCount, 0)
        label.Position = UDim2.new(0, 0, -(i - 1), 0)
        label.BackgroundTransparency = 1
        label.Text = text
        label.TextColor3 = Color3.fromRGB(220, 20, 20)
        label.FontFace = Font.new("rbxasset://fonts/families/SourceSansPro.json", Enum.FontWeight.Bold, Enum.FontStyle.Normal)
        label.TextSize = 28
        label.TextStrokeTransparency = 1
        label.TextTransparency = 1
        label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
        label.Parent = sliceFrame

        table.insert(labels, label)
        table.insert(sliceFrames, sliceFrame)
    end

    for _, lbl in ipairs(labels) do
        TweenService:Create(lbl, TweenInfo.new(0.35), {
            TextTransparency = 0,
            TextStrokeTransparency = 0.2
        }):Play()
    end

    local isAnimating = true
    task.spawn(function()
        local tweenInfo = TweenInfo.new(0.03, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
        local startTime = tick()
        while isAnimating and lineFrame.Parent do
            local elapsedTime = tick() - startTime
            for i, frame in ipairs(sliceFrames) do
                local waveOffsetX = math.sin(elapsedTime * 18 + (i * 0.5)) * 0.8
                if math.random(1, 100) <= 15 then
                    waveOffsetX = waveOffsetX + math.random(-1, 1)
                end
                TweenService:Create(frame, tweenInfo, {
                    Position = UDim2.new(0, waveOffsetX, (i - 1) * sliceHeightRatio, 0)
                }):Play()
            end
            task.wait(0.02)
        end
    end)

    task.delay(0.35 + 3.25, function()
        for _, lbl in ipairs(labels) do
            TweenService:Create(lbl, TweenInfo.new(0.75), {
                TextTransparency = 1,
                TextStrokeTransparency = 1
            }):Play()
        end
        task.delay(0.75, function()
            isAnimating = false
            lineFrame:Destroy()
        end)
    end)
end

local function triggerRandomLine(linesTable, customDelay)
    if #linesTable == 0 then return end
    local randomLine = linesTable[math.random(1, #linesTable)]
    local delayTime = (customDelay ~= nil) and customDelay or 1.1

    if delayTime > 0 then
        task.delay(delayTime, function()
            speakDialogue(randomLine)
        end)
    else
        speakDialogue(randomLine)
    end
end

-- Hàm Crossfade chính: Fade Out nhạc cũ VÀ Fade In nhạc mới CÙNG MỘT LÚC
local function crossfadeTo(targetSound, targetVolume, isLooped, duration)
    local fadeDuration = duration or FADE_TIME
    targetSound.Looped = (isLooped == true)

    if not targetSound.IsPlaying then
        targetSound.Volume = 0
        targetSound:Play()
    end

    -- Fade IN bài mới ngay lập tức
    TweenService:Create(targetSound, TweenInfo.new(fadeDuration, Enum.EasingStyle.Linear), {Volume = targetVolume}):Play()

    -- Fade OUT tất cả các bài còn lại song song cùng lúc
    for _, sound in ipairs(ALL_MUSIC_TRACKS) do
        if sound ~= targetSound and sound.IsPlaying then
            local tween = TweenService:Create(sound, TweenInfo.new(fadeDuration, Enum.EasingStyle.Linear), {Volume = 0})
            tween:Play()
            tween.Completed:Connect(function()
                if sound ~= targetSound and sound.Volume <= 0 then
                    sound:Stop()
                end
            end)
        end
    end
end

end

local function isMimicModel(obj)
    if not (obj and obj:IsA("Model")) then return false end
    local lowerName = obj.Name:lower()
    for _, keyword in ipairs(MIMIC_KEYWORDS) do
        if string.find(lowerName, keyword) then
            return true
        end
    end
    return false
end

local function getMimicCount()
    local count = 0
    for _, child in ipairs(workspace:GetChildren()) do
        if isMimicModel(child) then
            count += 1
        end
    end
    return count
end

workspace.ChildRemoved:Connect(function(child)

        end

        if isPhase3Active then

end)

local function isTargetPart(obj)
    if obj:IsA("BasePart") then
        local lowerName = obj.Name:lower()
        for _, targetName in ipairs(TARGET_PART_NAMES) do
            if string.find(lowerName, targetName) then
                return true
            end
        end
    end
    return false
end

local function fadeOutUI(screenGui, duration)
    if not screenGui then return end
    duration = duration or 0.5
    for _, obj in ipairs(screenGui:GetDescendants()) do
        if obj:IsA("TextLabel") then
            TweenService:Create(obj, TweenInfo.new(duration), {
                TextTransparency = 1,
                TextStrokeTransparency = 1
            }):Play()
        elseif obj:IsA("Frame") then
            TweenService:Create(obj, TweenInfo.new(duration), {
                BackgroundTransparency = 1
            }):Play()
        end
    end
    task.delay(duration + 0.1, function()
        if screenGui and screenGui.Parent then
            screenGui:Destroy()
        end
    end)
end

local function getAvailableMimics(phaseNum)
    local available = {}
    for _, url in ipairs(MIMIC_URLS) do
        if (phaseNum == 2 or phaseNum == 3) and url == DEER_URL then
            continue
        end

        if url == REBOUND_URL and reboundSpawned then
            continue
        end

        local isRecent = false
        for _, recentUrl in ipairs(recentMimics) do
            if url == recentUrl then
                isRecent = true
                break
            end
        end

        if not isRecent then
            local weight = 100

            if phaseNum == 2 then
                if url == SILENCE_URL then
                    weight = 40
                elseif url == BLITZ_URL or url == A120_URL then
                    weight = 30
                end
            elseif phaseNum == 3 then
                if url == SILENCE_URL then
                    weight = 20
                elseif url == BLITZ_URL or url == A120_URL then
                    weight = 15
                end
            end

            table.insert(available, {url = url, weight = weight})
        end
    end
    return available
end

local function spawnRandomMimic(phaseNum)
    local availableList = getAvailableMimics(phaseNum)
    if #availableList == 0 then return end

    local totalWeight = 0
    for _, item in ipairs(availableList) do
        totalWeight = totalWeight + item.weight
    end

    local randomVal = math.random(1, totalWeight)
    local currentSum = 0
    local selectedUrl = availableList[1].url

    for _, item in ipairs(availableList) do
        currentSum = currentSum + item.weight
        if randomVal <= currentSum then
            selectedUrl = item.url
            break
        end
    end

    if currentMusicState == "TENSE" or phaseNum == 1 then
        local tempPool = {}
        for _, line in ipairs(LINES_TENSE) do
            table.insert(tempPool, line)
        end
        if selectedUrl == A120_URL then
            table.insert(tempPool, LINE_A120_EXCLUSIVE)
        end
        triggerRandomLine(tempPool)
    end

    if selectedUrl == BLITZ_URL then
        flashSliceTextWhite()
    end

    if selectedUrl == REBOUND_URL then
        reboundSpawned = true
        nextDoorNoMimic = true
    end

    table.insert(recentMimics, selectedUrl)
    if #recentMimics > 2 then
        table.remove(recentMimics, 1)
    end

    task.spawn(function()
        pcall(function()
            loadstring(game:HttpGet(selectedUrl))()
        end)
    end)
end

    for _, descendant in ipairs(workspace:GetDescendants()) do
        if isTargetPart(descendant) then
            flashSliceTextWhite()
            break
        end
    end

    task.spawn(function()
        for i = 1, 3 do
            spawnRandomMimic(3)
            task.wait(0.2)
        end
    end)

    task.spawn(function()
        while isPhase3Active do
            local waitTime = math.random(13, 16)
            task.wait(waitTime)

            if not isPhase3Active then break end

            spawnRandomMimic(3)
        end
    end)

    local p3Debounce = false

    p3Connection = Rooms.ChildAdded:Connect(function(child)
        if not child:IsA("Model") or not isPhase3Active or p3Debounce then return end

        p3Debounce = true
        task.delay(0.3, function() p3Debounce = false end)

        phase3DoorsLeft -= 1
        if phase3DoorsLeft < 0 then phase3DoorsLeft = 0 end

        updateSliceText("B-" .. phase3DoorsLeft)

        if phase3DoorsLeft <= 0 then
            isPhase3Active = false

            if p3Connection then p3Connection:Disconnect() p3Connection = nil end
            if p3PartDetectorConnection then p3PartDetectorConnection:Disconnect() p3PartDetectorConnection = nil end

            fadeOutUI(screenGui, 0.5)

            task.spawn(function()
                while getMimicCount() > 0 do
                    task.wait(0.5)
                end

                local tween = TweenService:Create(phase3Sound, TweenInfo.new(FADE_TIME, Enum.EasingStyle.Linear), {Volume = 0})
                tween:Play()
                tween.Completed:Connect(function()
                    phase3Sound:Stop()
                end)

                triggerRandomLine(LINES_ENDING_FINAL)
            end)
        end
    end)
end

-- ======================================================
-- LOGIC PHASE 1
-- ======================================================

local function triggerPhase1Mimic()
    if isPhase1Ended then return end

    if nextDoorNoMimic then
        nextDoorNoMimic = false
        if getMimicCount() == 0 then
            playCalm2Music(true)
        end
        return
    end

    local roll = math.random(1, 100)
    if roll <= NO_MIMIC_CHANCE_P1 then
        if getMimicCount() == 0 then
            playCalm2Music(true)
        end
        return
    end

    playTenseMusic()
    spawnRandomMimic(1)
end

local function startPhase1DoorListener()
    local p1Debounce = false

    p1Connection = Rooms.ChildAdded:Connect(function(child)
        if not child:IsA("Model") or isPhase1Ended or p1Debounce then return end

        p1Debounce = true
        task.delay(0.3, function() p1Debounce = false end)

        doorsOpenedP1 += 1
        triggerPhase1Mimic()

        if doorsOpenedP1 >= MAX_DOORS_PHASE_1 then
            stopPhase1Music()

            if p1Connection then
                p1Connection:Disconnect()
                p1Connection = nil
            end

            task.delay(1.5, function()
                startPhase2()
            end)
        end
    end)
end

task.spawn(function()
    playCalm1Music(false)

    local startTick = tick()
    repeat task.wait(0.1) until calmSound.TimeLength > 0 or (tick() - startTick) > 3

    local duration = calmSound.TimeLength > 0 and calmSound.TimeLength or 15
    task.wait(duration)

    if not isPhase1Ended then
        triggerPhase1Mimic()
        startPhase1DoorListener()
    end
end)
